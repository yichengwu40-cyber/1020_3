ESP32 Arduino BLE + WS2812 控制程式碼

使用 NimBLE/BLE 藍牙函式庫與 Adafruit_NeoPixel 燈條控制。

#include <Adafruit_NeoPixel.h>
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

const int LDR_PIN = 39;     // 光感測器輸入 (ADC1_CH3)
const int POT_PIN = 34;     // 可變電阻輸入 (ADC1_CH6)
const int LED_PIN = 26;     // WS2812 資料接腳 (GPIO26)
const int NUM_LEDS = 3;     // 燈珠數量：3 顆

Adafruit_NeoPixel strip(NUM_LEDS, LED_PIN, NEO_GRB + NEO_KHZ800);

// BLE 服務與特徵值 UUID (對應網頁端設定)
#define SERVICE_UUID        "12345678-1234-1234-1234-000000000001"
#define INFO_UUID           "12345678-1234-1234-1234-000000000010"
#define DATA_UUID           "12345678-1234-1234-1234-000000000020"
#define COMMAND_UUID        "12345678-1234-1234-1234-000000000030"

BLECharacteristic *pDataCharacteristic;
bool deviceConnected = false;

class MyServerCallbacks: public BLEServerCallbacks {
    void onConnected(BLEServer* pServer) {
      deviceConnected = true;
    }
    void onDisconnected(BLEServer* pServer) {
      deviceConnected = false;
      pServer->startAdvertising(); // 重新廣播
    }
};

class MyCallbacks: public BLECharacteristicCallbacks {
    void onWrite(BLECharacteristic *pCharacteristic) {
      std::string rxValue = pCharacteristic->getValue();
      if (rxValue.length() > 0) {
        String cmd = String(rxValue.c_str());
        Serial.println("收到指令: " + cmd);
        // 您可以在此擴充接收網頁指令的邏輯
      }
    }
};

void setup() {
  Serial.begin(115200);
  strip.begin();
  strip.show();

  // 初始化 BLE
  BLEDevice::init("ESP32_WS2812_BLE");
  BLEServer *pServer = BLEDevice::createServer();
  pServer->setCallbacks(new MyServerCallbacks());

  BLEService *pService = pServer->createService(SERVICE_UUID);

  // 1. INFO 特徵值
  BLECharacteristic *pInfoCharacteristic = pService->createCharacteristic(
                        INFO_UUID,
                        BLECharacteristic::PROPERTY_READ
                      );
  pInfoCharacteristic->setValue("ESP32 WS2812 BLE Ready");

  // 2. DATA 特徵值 (Notify 用於傳送感測器 JSON)
  pDataCharacteristic = pService->createCharacteristic(
                        DATA_UUID,
                        BLECharacteristic::PROPERTY_NOTIFY
                      );
  pDataCharacteristic->addDescriptor(new BLE2902());

  // 3. COMMAND 特徵值 (Write 用於接收指令)
  BLECharacteristic *pCommandCharacteristic = pService->createCharacteristic(
                        COMMAND_UUID,
                        BLECharacteristic::PROPERTY_WRITE
                      );
  pCommandCharacteristic->setCallbacks(new MyCallbacks());

  pService->start();
  BLEAdvertising *pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  pAdvertising->setScanResponse(true);
  pAdvertising->setMinPreferred(0x06); 
  BLEDevice::startAdvertising();
  Serial.println("BLE 藍牙就緒，等待連線...");
}

void loop() {
  // 1. 讀取感測器數值 (0 ~ 4095)
  int ldrValue = analogRead(LDR_PIN);
  int potValue = analogRead(POT_PIN);

  // 2. 映射邏輯：環境越暗 (LDR 越小)，WS2812 亮度越大 (0-255)
  int brightness = map(ldrValue, 0, 4095, 255, 0);
  brightness = constrain(brightness, 0, 255);

  strip.setBrightness(brightness);
  strip.setPixelColor(0, strip.Color(255, 50, 150));
  strip.setPixelColor(1, strip.Color(100, 50, 255));
  strip.setPixelColor(2, strip.Color(0, 220, 255));
  strip.show();

  // 3. 若 BLE 已連線，組合 JSON 並透過 Notification 傳送給網頁
  if (deviceConnected) {
    String jsonString = "{\"ldr\":" + String(ldrValue) + 
                        ",\"pot\":" + String(potValue) + 
                        ",\"led\":" + String(brightness) + "}";
    pDataCharacteristic->setValue(jsonString.c_str());
    pDataCharacteristic->notify();
  }

  delay(100);
}
